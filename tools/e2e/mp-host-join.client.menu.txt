# The joining player's path through the real lobby UI: landing, join setup, connect, the read-only
# match details, ready. The match is driven by this peer's probe; the script leaves afterwards.
wait 60
activate ButtonMainToMultiplayer
wait 15
assert_substate Landing
settext TextMultiplayerName Joiner
activate ButtonMultiplayerJoinGame
wait 20
assert_substate JoinSetup
settext TextJoinAddress 127.0.0.1
settext TextJoinPort {PORT}
wait 15
activate ButtonMultiplayerConnect
wait_connected 2
wait 25
assert_substate Lobby
dump_lobby
activate ButtonLobbyOptions
wait 15
assert_substate HostOptions
assert_label LabelHostOptionsTitle M A T C H   D E T A I L S
activate TabHostPageRules
wait 20
assert_visible CollectionBoxHostPageRules 1
assert_enabled ButtonHostOptApply 0
dump_host_options
activate ButtonHostOptBack
wait 15
assert_substate Lobby
activate ButtonMultiplayerReady
wait 15
dump_lobby
wait_file {PROBE_DIR}/done.json 240
wait 40
assert_substate Lobby
dump_lobby
wait 60
activate ButtonMultiplayerLeave
wait 40
assert_substate Landing
wait 20
exit
